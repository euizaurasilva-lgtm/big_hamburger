    function openTableModal(tableId) {
      const t = tables.find(x => x.id === tableId);
      const action = prompt(Mesa ${tableId} (${t.occupied ? 'OCUPADA - ' + formatMoney(t.total) : 'LIVRE'})\n\nDigite:\n1 - Adicionar Consumo\n2 - Fechar Conta);
      if (action === "1") {
        const val = parseFloat(prompt("Valor do item lançado (R$):", "35.00"));
        if (val) {
          t.occupied = true;
          t.total += val;
          renderTables();
        }
      } else if (action === "2" && t.occupied) {
        alert(Conta da Mesa ${tableId} encerrada no valor de ${formatMoney(t.total)}!);
        t.occupied = false;
        t.total = 0;
        renderTables();
      }
    }
